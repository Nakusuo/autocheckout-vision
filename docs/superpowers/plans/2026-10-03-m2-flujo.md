# M2 Flujo con simuladores: plan de implementación

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Objetivo:** recorrer el flujo W1 → W4 de punta a punta en el navegador. Se usan una bandeja simulada (en lugar de la cámara), la balanza simulada y un botón provisional de «simular pago».

**Arquitectura:** `kiosk-api` gana cuatro piezas: SQLite para el catálogo, `SimulatedTray` (el detector falso), `KioskService` (que une las piezas de M0, los temporizadores de 10 s y 120 s y las alertas) y una app FastAPI con REST, WebSocket y rutas `/sim/*`. `kiosk-ui` es React + Vite + TypeScript con tres vistas: el quiosco (`/`), el panel de simulación (`/sim`) y el panel de caja (`/caja`).

**Stack:** Python 3.12, FastAPI, uvicorn, sqlite3 (stdlib), pytest, httpx · Node 22+, Vite, React 18, TypeScript, Vitest y Testing Library.

**Spec:** [`docs/superpowers/specs/2026-10-03-autocheckout-design.md`](../specs/2026-10-03-autocheckout-design.md) · Base: plan M0 (`2026-10-03-m0-base.md`)

## Restricciones globales

- Todas las del plan M0 siguen vigentes (±5 g, de 1 a 3 productos, 3 cuadros, precios en céntimos `int`, sin coautor y textos en español).
- Temporizadores: vuelta a W1 **10 s** después de PAID (3.4.4.4); cobro cancelado y vuelta a W3 a los **120 s** en PAYING.
- Texto exacto de W1: `[AUTO-CHECKOUT] Coloca tus productos` y botón `INICIAR` (3.4.3.1).
- Botones exactos: `INICIAR`, `CONFIRMAR`, `CORREGIR`, `PAGAR`, `SOLICITAR AYUDA`.
- Alertas exactas: `Revise su bandeja`, `Para más de 3 productos use la caja`, `Hay un producto que no reconocemos`.
- Moneda: `S/ 2.50` (prefijo `S/`, espacio, 2 decimales con punto).
- Formato vertical 9:16, pensado para 1080×1920 (RNF-06). Objetivos táctiles de **al menos 64 px de alto** a 1080 de ancho.
- **Marca genérica «AutoCaja»**: no usar el nombre, el logo ni el rojo/amarillo de OXXO.
- Los endpoints de FastAPI son `async def`: así comparten hilo con el bucle de `tick()` y no hay condiciones de carrera.
- Las rutas `/sim/*` solo existen si se pasan los simuladores a `create_app`.
- Commits en `feature/m2-flujo`, sin Co-Authored-By.

```python
import sqlite3
from contextlib import closing
from pathlib import Path

from kiosk.catalog import Catalog, Product

SCHEMA = """
CREATE TABLE IF NOT EXISTS products (
    sku TEXT PRIMARY KEY,
    name TEXT NOT NULL,
    price_cents INTEGER NOT NULL CHECK (price_cents >= 0),
    weight_g REAL NOT NULL CHECK (weight_g > 0)
)
"""


def init_db(db_path: Path, seed_csv: Path) -> None:
    rows = [
        (p.sku, p.name, p.price_cents, p.weight_g) for p in Catalog.from_csv(seed_csv).all()
    ]
    with closing(sqlite3.connect(db_path)) as conn, conn:
        conn.execute(SCHEMA)
        conn.executemany("INSERT OR REPLACE INTO products VALUES (?, ?, ?, ?)", rows)


def load_catalog(db_path: Path) -> Catalog:
    with closing(sqlite3.connect(db_path)) as conn:
        rows = conn.execute("SELECT sku, name, price_cents, weight_g FROM products").fetchall()
    return Catalog([Product(*row) for row in rows])
```

```python
from typing import Protocol, runtime_checkable

from kiosk.catalog import Catalog
from kiosk.scale import SimulatedScale


@runtime_checkable
class Detector(Protocol):
    def detect(self) -> list[str]: ...


class SimulatedTray:
    """Bandeja simulada: hace de cámara (detector) y carga la balanza simulada."""

    def __init__(self, catalog: Catalog, scale: SimulatedScale):
        self._catalog = catalog
        self._scale = scale
        self._skus: list[str] = []

    def place(self, skus: list[str]) -> None:
        products = []
        for sku in skus:
            product = self._catalog.get(sku)
            if product is None:
                raise ValueError(f"Producto desconocido: {sku}")
            products.append(product)
        self._skus = list(skus)
        self._scale.set_load(sum(p.weight_g for p in products))

    def detect(self) -> list[str]:
        return list(self._skus)
```

```python
import logging
import time
from collections.abc import Callable

from kiosk.catalog import Catalog
from kiosk.checkout import MAX_ITEMS, Checkout, State
from kiosk.fusion import build_cart, check_weight
from kiosk.scale import ScaleReader
from kiosk.stability import StabilityFilter
from kiosk.tray import Detector

log = logging.getLogger(__name__)

RESET_AFTER_S = 10.0
PAYMENT_TIMEOUT_S = 120.0

SCREEN = {
    State.IDLE: "W1",
    State.SCANNING: "W2",
    State.CONFIRMING: "W3",
    State.PAYING: "QR",
    State.PAID: "W4",
    State.OUT_OF_SERVICE: "FUERA_DE_SERVICIO",
}


class KioskService:
    def __init__(
        self,
        catalog: Catalog,
        detector: Detector,
        scale: ScaleReader,
        clock: Callable[[], float] = time.monotonic,
    ):
        self.catalog = catalog
        self._detector = detector
        self._scale = scale
        self._clock = clock
        self.checkout = Checkout()
        self._stability = StabilityFilter()
        self._entered_at = clock()
        self.help_requests: list[dict] = []
        self._next_help_id = 1

    def _do(self, action: Callable[[], None]) -> None:
        before = self.checkout.state
        action()
        if self.checkout.state is not before:
            self._entered_at = self._clock()
            if self.checkout.state is State.SCANNING:
                self._stability.reset()

    def start(self) -> None:
        self._do(self.checkout.start)

    def confirm(self) -> None:
        self._do(self.checkout.confirm)

    def correct(self) -> None:
        self._do(self.checkout.correct)

    def pay(self) -> None:
        self._do(self.checkout.pay)

    def simulate_payment(self) -> None:
        self._do(self.checkout.payment_confirmed)

    def recover(self) -> None:
        self._do(self.checkout.recover)

    def tick(self) -> None:
        state = self.checkout.state
        elapsed = self._clock() - self._entered_at
        if state is State.PAID and elapsed >= RESET_AFTER_S:
            self._do(self.checkout.reset)
            return
        if state is State.PAYING and elapsed >= PAYMENT_TIMEOUT_S:
            self._do(self.checkout.payment_timeout)
            return
        if state not in (State.SCANNING, State.CONFIRMING):
            return
        try:
            skus = self._detector.detect()
            grams = self._scale.read()
        except Exception:
            log.exception("Fallo de sensor; quiosco fuera de servicio")
            self._do(self.checkout.fault)
            return
        stable = self._stability.push(skus)
        if stable is None:
            return
        cart = build_cart(stable, self.catalog)
        if state is State.CONFIRMING and cart != self.checkout.cart:
            self._do(self.checkout.correct)
        self.checkout.update(cart, check_weight(cart.theoretical_weight_g, grams))

    def request_help(self) -> dict:
        req = {
            "id": self._next_help_id,
            "screen": SCREEN[self.checkout.state],
            "at": self._clock(),
        }
        self._next_help_id += 1
        self.help_requests.append(req)
        return req

    def resolve_help(self, help_id: int) -> None:
        self.help_requests = [r for r in self.help_requests if r["id"] != help_id]

    def _alert(self) -> str | None:
        cart, weight = self.checkout.cart, self.checkout.weight
        if cart is None:
            return None
        if cart.item_count > MAX_ITEMS:
            return "Para más de 3 productos use la caja"
        if cart.unknown_skus:
            return "Hay un producto que no reconocemos"
        if weight is not None and not weight.ok:
            return "Revise su bandeja"
        return None

    def snapshot(self) -> dict:
        c = self.checkout
        cart = c.cart
        return {
            "state": c.state.value,
            "screen": SCREEN[c.state],
            "can_pay": c.can_pay,
            "lines": [
                {
                    "sku": line.sku,
                    "name": line.name,
                    "quantity": line.quantity,
                    "unit_price_cents": line.unit_price_cents,
                }
                for line in (cart.lines if cart else ())
            ],
            "total_cents": cart.total_cents if cart else 0,
            "item_count": cart.item_count if cart else 0,
            "weight": {"ok": c.weight.ok, "diff_g": c.weight.diff_g} if c.weight else None,
            "alert": self._alert(),
            "seconds_in_state": round(self._clock() - self._entered_at, 1),
            "help_pending": bool(self.help_requests),
        }
```

```python
import asyncio
from collections.abc import Callable
from contextlib import asynccontextmanager

from fastapi import FastAPI, HTTPException, WebSocket, WebSocketDisconnect
from pydantic import BaseModel

from kiosk.checkout import TransitionError
from kiosk.scale import SimulatedScale
from kiosk.service import KioskService
from kiosk.tray import SimulatedTray

TICK_S = 0.2


class TrayIn(BaseModel):
    skus: list[str]


class TrapIn(BaseModel):
    offset_g: float


def create_app(
    service: KioskService,
    tray: SimulatedTray | None = None,
    scale: SimulatedScale | None = None,
    run_loop: bool = True,
) -> FastAPI:
    async def tick_loop() -> None:
        while True:
            service.tick()
            await asyncio.sleep(TICK_S)

    @asynccontextmanager
    async def lifespan(_: FastAPI):
        task = asyncio.create_task(tick_loop()) if run_loop else None
        yield
        if task:
            task.cancel()

    app = FastAPI(title="kiosk-api", lifespan=lifespan)

    def act(action: Callable[[], None]) -> dict:
        try:
            action()
        except TransitionError as e:
            raise HTTPException(status_code=409, detail=str(e)) from e
        return service.snapshot()

    @app.get("/api/state")
    async def state() -> dict:
        return service.snapshot()

    @app.post("/api/start")
    async def start() -> dict:
        return act(service.start)

    @app.post("/api/confirm")
    async def confirm() -> dict:
        return act(service.confirm)

    @app.post("/api/correct")
    async def correct() -> dict:
        return act(service.correct)

    @app.post("/api/pay")
    async def pay() -> dict:
        return act(service.pay)

    @app.post("/api/recover")
    async def recover() -> dict:
        return act(service.recover)

    @app.post("/api/help")
    async def request_help() -> dict:
        service.request_help()
        return service.snapshot()

    @app.get("/api/help")
    async def list_help() -> list[dict]:
        return service.help_requests

    @app.delete("/api/help/{help_id}")
    async def resolve_help(help_id: int) -> list[dict]:
        service.resolve_help(help_id)
        return service.help_requests

    @app.websocket("/ws")
    async def ws(websocket: WebSocket) -> None:
        await websocket.accept()
        try:
            while True:
                await websocket.send_json(service.snapshot())
                await asyncio.sleep(TICK_S)
        except (WebSocketDisconnect, RuntimeError):
            pass

    if tray is not None and scale is not None:

        @app.get("/sim/catalog")
        async def sim_catalog() -> list[dict]:
            return [
                {"sku": p.sku, "name": p.name, "price_cents": p.price_cents, "weight_g": p.weight_g}
                for p in service.catalog.all()
            ]

        @app.post("/sim/tray")
        async def sim_tray(body: TrayIn) -> dict:
            try:
                tray.place(body.skus)
            except ValueError as e:
                raise HTTPException(status_code=422, detail=str(e)) from e
            return service.snapshot()

        @app.post("/sim/trap")
        async def sim_trap(body: TrapIn) -> dict:
            scale.set_trap(body.offset_g)
            return service.snapshot()

        @app.post("/sim/pay")
        async def sim_pay() -> dict:
            return act(service.simulate_payment)

    return app
```
