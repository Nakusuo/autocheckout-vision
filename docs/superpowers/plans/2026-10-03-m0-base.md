**Arquitectura:** paquete Python `kiosk` dentro de `kiosk-api/src/kiosk/`. Contiene funciones puras y clases pequeñas: catálogo, carrito y fusión de peso, filtro de estabilidad, máquina de estados y balanza simulada. FastAPI, WebSocket y SQLite llegan en M2 y se apoyan en estas piezas.

**Spec:** [`docs/superpowers/specs/2026-10-03-autocheckout-design.md`](../specs/2026-10-03-autocheckout-design.md)

## Restricciones globales

- Tolerancia de peso: `|real − teórico| ≤ 5 g`. El límite exacto de 5,0 g **se acepta** (3.4.2.3).
- Máximo **3** productos por transacción; mínimo **1** producto válido (3.4.1.2 y 3.4.1.4).
- Lista estable: el mismo conjunto de SKU en **3** cuadros seguidos.
- Precios en **céntimos enteros** (`int`), nunca `float`.
- Pesos en gramos (`float`), redondeados a 1 decimal en las lecturas.
- Python ≥ 3.12. Sin dependencias de producción en M0.
- Commits en `feature/*`, sin línea de coautor; se unen a `develop` con `merge --no-ff`.
- Mensajes, errores y nombres visibles al usuario en español.

```python
import csv
from dataclasses import dataclass
from pathlib import Path


@dataclass(frozen=True)
class Product:
    sku: str
    name: str
    price_cents: int
    weight_g: float


class Catalog:
    def __init__(self, products: list[Product]):
        self._by_sku = {p.sku: p for p in products}

    @classmethod
    def from_csv(cls, path: Path) -> "Catalog":
        with path.open(encoding="utf-8", newline="") as f:
            return cls(
                [
                    Product(r["sku"], r["name"], int(r["price_cents"]), float(r["weight_g"]))
                    for r in csv.DictReader(f)
                ]
            )

    def get(self, sku: str) -> Product | None:
        return self._by_sku.get(sku)
```

```python
from collections import Counter
from dataclasses import dataclass

from kiosk.catalog import Catalog

TOLERANCE_G = 5.0


@dataclass(frozen=True)
class CartLine:
    sku: str
    name: str
    quantity: int
    unit_price_cents: int


@dataclass(frozen=True)
class Cart:
    lines: tuple[CartLine, ...]
    theoretical_weight_g: float
    total_cents: int
    unknown_skus: tuple[str, ...]

    @property
    def item_count(self) -> int:
        return sum(line.quantity for line in self.lines)


def build_cart(skus: list[str], catalog: Catalog) -> Cart:
    lines: list[CartLine] = []
    unknown: list[str] = []
    weight = 0.0
    total = 0
    for sku, qty in sorted(Counter(skus).items()):
        product = catalog.get(sku)
        if product is None:
            unknown.extend([sku] * qty)
            continue
        lines.append(CartLine(sku, product.name, qty, product.price_cents))
        weight += product.weight_g * qty
        total += product.price_cents * qty
    return Cart(tuple(lines), round(weight, 1), total, tuple(unknown))


@dataclass(frozen=True)
class WeightCheck:
    ok: bool
    diff_g: float


def check_weight(
    theoretical_g: float, measured_g: float, tolerance_g: float = TOLERANCE_G
) -> WeightCheck:
    diff = round(abs(measured_g - theoretical_g), 1)
    return WeightCheck(ok=diff <= tolerance_g, diff_g=diff)
```

Nota: `round(..., 1)` antes de comparar evita que errores de coma flotante (p. ej. `105.0 − 100.0`) rechacen el límite exacto.

```python
from collections import Counter


class StabilityFilter:
    """Da por buena una detección cuando se repite igual en N cuadros seguidos."""

    def __init__(self, frames_required: int = 3):
        self._required = frames_required
        self._last: Counter[str] | None = None
        self._count = 0

    def push(self, skus: list[str]) -> list[str] | None:
        current = Counter(skus)
        if current == self._last:
            self._count += 1
        else:
            self._last = current
            self._count = 1
        if self._count >= self._required:
            return sorted(current.elements())
        return None

    def reset(self) -> None:
        self._last = None
        self._count = 0
```

```python
from enum import Enum

from kiosk.fusion import Cart, WeightCheck

MAX_ITEMS = 3


class State(str, Enum):
    IDLE = "IDLE"  # W1
    SCANNING = "SCANNING"  # W2
    CONFIRMING = "CONFIRMING"  # W3
    PAYING = "PAYING"  # QR
    PAID = "PAID"  # W4
    OUT_OF_SERVICE = "OUT_OF_SERVICE"


class TransitionError(Exception):
    pass


class Checkout:
    def __init__(self) -> None:
        self.state = State.IDLE
        self.cart: Cart | None = None
        self.weight: WeightCheck | None = None

    def _require(self, *states: State) -> None:
        if self.state not in states:
            raise TransitionError(f"Acción no permitida en el estado {self.state.value}")

    def _clear(self) -> None:
        self.cart = None
        self.weight = None

    @property
    def can_pay(self) -> bool:
        return self.state is State.CONFIRMING and self.weight is not None and self.weight.ok

    def start(self) -> None:
        self._require(State.IDLE)
        self.state = State.SCANNING

    def update(self, cart: Cart, weight: WeightCheck) -> None:
        self._require(State.SCANNING, State.CONFIRMING)
        self.cart = cart
        self.weight = weight

    def confirm(self) -> None:
        self._require(State.SCANNING)
        cart = self.cart
        if cart is None or cart.unknown_skus or not 1 <= cart.item_count <= MAX_ITEMS:
            raise TransitionError("Se necesitan de 1 a 3 productos reconocidos")
        self.state = State.CONFIRMING

    def correct(self) -> None:
        self._require(State.CONFIRMING)
        self.state = State.SCANNING

    def pay(self) -> None:
        self._require(State.CONFIRMING)
        if not self.can_pay:
            raise TransitionError("Revise su bandeja")
        self.state = State.PAYING

    def payment_confirmed(self) -> None:
        self._require(State.PAYING)
        self.state = State.PAID

    def payment_timeout(self) -> None:
        self._require(State.PAYING)
        self.state = State.CONFIRMING

    def reset(self) -> None:
        self._require(State.PAID)
        self._clear()
        self.state = State.IDLE

    def fault(self) -> None:
        self.state = State.OUT_OF_SERVICE

    def recover(self) -> None:
        self._require(State.OUT_OF_SERVICE)
        self._clear()
        self.state = State.IDLE
```

```python
import random
from typing import Protocol, runtime_checkable


@runtime_checkable
class ScaleReader(Protocol):
    def read(self) -> float: ...


class SimulatedScale:
    """Balanza falsa: peso cargado ± ruido, más un desvío opcional («modo trampa»)."""

    def __init__(self, noise_g: float = 2.0, seed: int | None = None):
        self._rng = random.Random(seed)  # nosec B311 — simulación, no criptografía
        self._noise = noise_g
        self._load = 0.0
        self._offset = 0.0

    def set_load(self, grams: float) -> None:
        self._load = grams

    def set_trap(self, offset_g: float) -> None:
        self._offset = offset_g

    def read(self) -> float:
        grams = self._load + self._offset + self._rng.uniform(-self._noise, self._noise)
        return round(max(grams, 0.0), 1)
```
